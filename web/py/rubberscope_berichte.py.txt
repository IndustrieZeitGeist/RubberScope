def _p(s, style, markup=False):
    from reportlab.platypus import Paragraph
    return Paragraph(str(s) if markup else escape(str(s)), style)


def _tab(rows, widths, sty):
    from reportlab.platypus import LongTable
    from reportlab.lib import colors
    from reportlab.platypus import TableStyle
    data = [[_p(v, sty['cb'] if i == 0 else sty['cell']) for v in row] for i, row in enumerate(rows)]
    tbl = LongTable(data, colWidths=widths, repeatRows=1, hAlign='LEFT')
    tbl.setStyle(TableStyle([
        ('VALIGN',(0,0),(-1,-1),'TOP'), ('ROWBACKGROUNDS',(0,1),(-1,-1),[colors.white,colors.HexColor('#F2F6F9')]),
        ('BACKGROUND',(0,0),(-1,0),colors.HexColor('#DDECF2')), ('LINEBELOW',(0,0),(-1,-1),0.25,colors.HexColor('#C9D8DE')),
        ('TOPPADDING',(0,0),(-1,-1),5),('BOTTOMPADDING',(0,0),(-1,-1),5),
    ]))
    return tbl


def _build(out, title, story, sty):
    from reportlab.platypus import SimpleDocTemplate, Spacer
    from reportlab.lib.pagesizes import A4
    out = Path(out)
    out.parent.mkdir(parents=True, exist_ok=True)
    doc = SimpleDocTemplate(str(out), pagesize=A4, topMargin=43, bottomMargin=45,
                            leftMargin=43, rightMargin=43, title=title,
                            subject='RubberScope ' + eng.VERSION)
    def foot(canvas, document):
        canvas.setFont(sty['reg'], 7)
        canvas.drawString(43, 22, 'RubberScope ' + eng.VERSION + ' | explorative Bildanalyse, nicht normzertifiziert')
        canvas.drawRightString(A4[0]-43, 22, 'Seite ' + str(document.page))
    doc.build([_p(title, sty['title']), Spacer(1,8)] + story, onFirstPage=foot, onLaterPages=foot)
    return out

def _beleg(*source_ids):
    return ' Beleg: ' + ', '.join('[' + s + ']' for s in source_ids) + '.'


def _statfmt(value):
    if value is None: return '-'
    try: return f'{float(value):.5g}'
    except (ValueError, TypeError): return str(value)

def _refmask(path):
    arr = np.asarray(Image.open(path).convert('L'))
    if not arr.size:
        raise ValueError('Leere Referenzmaske: ' + str(path))
    if int(arr.max()) <= 1:
        return arr > 0
    return arr >= 127


def validate_real_reference_set(root, folder, sensitivity=1.0, min_area=16, algorithms=('rissoptimiert','erweitert')):
    """Prueft fremdannotierte reale Bilder. Kein Nachkalibrieren am Referenzset.

    Ordnerlayout: bilder/<name>.bmp und masken/<name>.bmp (alternativ *_mask.png / *_maske.png).
    Alternativ (z. B. RubberScope-Testbilder): Aufnahmen direkt im gewaehlten Ordner, Masken in masken/.
    Leere (schwarze) Masken gelten als rissfreie Referenz und werden nur auf Fehlalarme geprueft.
    """
    root, folder = Path(root), Path(folder)
    folder.mkdir(parents=True, exist_ok=True)
    if root.name.lower() == 'masken' and (root.parent / 'masken').is_dir():
        root = root.parent
    imgs, masks = root/'bilder', root/'masken'
    if not masks.is_dir():
        raise FileNotFoundError('Erforderlich: Ordner masken/ (Sollmasken) und die Aufnahmen in bilder/ oder direkt '
                                'im gewaehlten Ordner. Bitte den Ordner waehlen, der masken/ enthaelt.')
    strict = imgs.is_dir()
    if not strict:
        imgs = root
    found = sorted(p for p in imgs.iterdir() if p.is_file() and p.suffix.lower() in eng.EXTS)
    # gleiche Aufnahme als TIF und PNG nur einmal werten (TIF bevorzugt)
    by_stem = {}
    for p in found:
        q = by_stem.get(p.stem.lower())
        if q is None or (p.suffix.lower() in ('.tif', '.tiff') and q.suffix.lower() not in ('.tif', '.tiff')):
            by_stem[p.stem.lower()] = p
    images = sorted(by_stem.values())
    if not images:
        raise ValueError('Keine Aufnahmen gefunden (erwartet in bilder/ oder direkt im gewaehlten Ordner).')
    rows, skipped = [], []
    for ip in images:
        candidates = [masks / (ip.stem + sx) for sx in ('.bmp','.png','.tif','_mask.png','_maske.png','_gt.png','.jpg')]
        mp = next((p for p in candidates if p.is_file()),None)
        if mp is None and not strict:
            skipped.append(ip.name)
            continue
        if mp is None:
            raise FileNotFoundError('Fehlende Ground-Truth-Maske zu ' + ip.name)
        rgb, gray, _ = eng.read_img(ip)
        gt = _refmask(mp)
        if gray.shape != gt.shape:
            raise ValueError(f'Groessenabweichung ohne automatische Skalierung: {ip.name} {gray.shape} vs. {mp.name} {gt.shape}')
        if gt.mean() > 0.75:
            raise ValueError('Die Referenzmaske enthaelt >75% positive Pixel: Vorzeichen/Maskenkodierung pruefen: ' + mp.name)
        for alg in algorithms:
            pred = eng.make_mask(gray,'riss','dunkel',sensitivity,min_area,alg).astype(bool)
            TP = int(np.count_nonzero(pred & gt)); FP = int(np.count_nonzero(pred & ~gt))
            FN = int(np.count_nonzero(~pred & gt)); TN = int(np.count_nonzero(~pred & ~gt))
            precision = TP/(TP+FP) if (TP+FP) else (1.0 if not gt.any() else 0.0)
            recall = TP/(TP+FN) if (TP+FN) else (1.0 if not pred.any() else 0.0)
            f1 = 2*TP/(2*TP+FP+FN) if (2*TP+FP+FN) else 1.0
            iou = TP/(TP+FP+FN) if (TP+FP+FN) else 1.0
            Lgt = eng.skeleton_length(__import__('skimage.morphology',fromlist=['skeletonize']).skeletonize(gt))
            Lpred = eng.skeleton_length(__import__('skimage.morphology',fromlist=['skeletonize']).skeletonize(pred))
            lr = 100*(Lpred-Lgt)/Lgt if Lgt > 0 else None
            overlay = rgb.copy()
            overlay[gt & pred] = [55,170,95]
            overlay[pred & ~gt] = [235,75,75]
            overlay[gt & ~pred] = [60,125,235]
            out_overlay=folder/f'{ip.stem}_{alg}_referenzvergleich.png'
            Image.fromarray(overlay).save(out_overlay)
            rows.append(dict(bild=ip.name,maske=mp.name,sha256=eng.file_sha256(ip),maske_sha256=eng.file_sha256(mp),
                             verfahren=alg,k=sensitivity,A_min_px2=min_area,TP=TP,FP=FP,FN=FN,TN=TN,
                             precision=round(precision,5),recall=round(recall,5),f1=round(f1,5),iou=round(iou,5),
                             laengenfehler_prozent=round(lr,3) if lr is not None else None,
                             referenz_risspixel=int(gt.sum()),test_risspixel=int(pred.sum()),
                             rissfrei=not bool(gt.any()),
                             groesse=f'{gray.shape[1]}x{gray.shape[0]}',overlay=out_overlay.name))
    if not rows:
        raise FileNotFoundError('Zu keiner Aufnahme wurde eine passende Maske in masken/ gefunden '
                                '(Maske gleich benannt oder mit Zusatz _maske / _mask).')
    with (folder/'validierung_echte_bilder.csv').open('w',encoding='utf-8-sig',newline='') as f:
        writer=csv.DictWriter(f,fieldnames=list(rows[0]),delimiter=';');writer.writeheader();writer.writerows(rows)
    meta_path = root/'QUELLE_UND_LIZENZ.json'
    provenance = json.loads(meta_path.read_text(encoding='utf-8')) if meta_path.exists() else {'status':'Datenherkunft nicht durch Manifest belegt; bitte manuell dokumentieren.'}
    data={'type':'real_fremdmaterial', 'scope':'Fremdmaterial-Kontrolltest; KEINE Validierung an ozongealtertem Elastomer',
          'documentation': 'Manuelle Pixelannotationen als Ground Truth; k fix vor Sichtung; Bilder sind statistische Einheiten nur im Datensatz.',
          'source':provenance,'software':eng.environment_info(),'records':rows,'ohne_maske_uebersprungen':skipped,
          'settings':{'sensitivity':sensitivity,'min_area_px2':min_area,'algorithms':list(algorithms)}}
    (folder/'validierung_echte_bilder.json').write_text(json.dumps(data,ensure_ascii=False,indent=2,default=str),encoding='utf-8')
    return data


def validierungsprotokoll(validation, folder, suffix='synthetisch'):
    """Separates PDF mit F1, Precision, Recall, TP/FP/FN und Einzelergebnissen."""
    folder=Path(folder);folder.mkdir(parents=True,exist_ok=True)
    sty=_styles();story=[]
    P=lambda t: story.append(_p(t,sty['normal']))
    H=lambda t: story.append(_p(t,sty['h']))
    isreal = validation.get('type') == 'real_fremdmaterial'
    H('1. Untersuchungsgegenstand')
    if isreal:
        P('Reale, extern annotierte Rissbilder. Ueberprueft wird die technische Uebereinstimmung der berechneten Segmentierungsmaske mit manuell markierten Bildpixeln, nicht das Verhalten einer ozongealterten Elastomerprobe.')
        src=validation.get('source',{})

    else:
        P(validation.get('method','Sechs synthetische Bilder mit bekannten Pixelmasken.'))

        grid = validation.get('k_grid')
        if grid:
            P('Gepruefte Empfindlichkeiten k: ' + '; '.join(f'{a}: ' + ', '.join(f'{x:g}' for x in g) for a, g in grid.items()) + '.' + _beleg('R08'))
            edge = [a for a, e in (validation.get('k_at_grid_edge') or {}).items() if e]
            if edge:
                P('HINWEIS: Das gewaehlte k liegt am Rand des geprueften Rasters fuer ' + ', '.join(edge) + '. Das Optimum kann ausserhalb liegen; Raster erweitern.')
        if validation.get('method_params') is not None:
            ch = validation.get('method_params_changed') or []
            P('Methodenparameter: Satz ' + str(validation.get('method_params_hash')) + '; ' + ('alle Werte = Programmstandard.' if not ch else 'vom Standard abweichend: ' + ', '.join(f"{k} = {eng.fmt_param(validation['method_params'][k])}" for k in ch) + '.') + _beleg('R09'))
        records=validation.get('detail',[])
    H('2. Metriken und Berechnung')
    P('TP: Uebereinstimmende Risspixel; FP: automatisch segmentierte Pixel ohne Ground-Truth-Riss; FN: nicht erkannte Ground-Truth-Risspixel; TN: gemeinsam als Hintergrund klassifizierte Pixel.')
    P('Precision = TP/(TP+FP); Recall = TP/(TP+FN); F1 = 2TP/(2TP+FP+FN); IoU = TP/(TP+FP+FN). Bei leeren Masken gelten programmspezifische Sonderfallregeln.' + _beleg('Q14','R08'))
    P('Relativer Laengenfehler = (L_Algorithmus - L_Referenz)/L_Referenz * 100%. Definition des programmspezifischen Fehlermasses; Laengen im Pixelmassstab, ohne Kalibrierung keine Millimeterwerte.' + _beleg('R05','R08'))

    if not isreal:

        story.append(_tab([['Algorithmus','k gewaehlt','F1 Entwicklung','F1 Holdout','Laengenfehler Holdout']] +
                          [[key,str(val.get('sensitivity')),str(val.get('development_f1')),str(val.get('holdout_f1')),
